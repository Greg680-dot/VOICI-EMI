// Filter Buttons Interaction
  document.querySelectorAll('#filter-format button').forEach(button => {
    button.addEventListener('click', () => {
      document.querySelectorAll('#filter-format button').forEach(b => {
        b.classList.remove('bg-primary', 'text-on-primary', 'active');
        b.classList.add('bg-surface-container-low', 'text-on-surface-variant');
      });
      button.classList.add('bg-primary', 'text-on-primary', 'active');
      button.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
      showToast('Filtre appliqué', 'Mise à jour des événements en cours...');
    });
  });

  // Timezone Switcher Interaction
  document.querySelectorAll('#filter-timezone button').forEach(button => {
    button.addEventListener('click', () => {
      document.querySelectorAll('#filter-timezone button').forEach(b => {
        b.classList.remove('bg-secondary', 'text-on-secondary', 'active');
        b.classList.add('bg-surface-container-low', 'text-on-surface-variant');
      });
      button.classList.add('bg-secondary', 'text-on-secondary', 'active');
      button.classList.remove('bg-surface-container-low', 'text-on-surface-variant');
      showToast('Fuseau synchronisé', 'Horaires convertis pour ' + button.innerText.trim());
    });
  });

  // 1-Click Instant Register Handler
  function handleInstantRegister(btn) {
    const originalText = btn.innerHTML;
    btn.innerHTML = `<span class="material-symbols-outlined animate-spin text-[20px]">refresh</span> <span>Génération du pass...</span>`;
    btn.disabled = true;

    setTimeout(() => {
      btn.classList.remove('bg-primary');
      btn.classList.add('bg-secondary');
      btn.innerHTML = `<span class="material-symbols-outlined text-[20px]">check</span> <span>Place Réservée !</span>`;
      showToast('Accès Masterclasse Accordé !', 'Votre pass numérique chiffré sans mot de passe est actif.');
    }, 800);
  }

  // Candidacy Form Handler
  function handleCandidacySubmit(e) {
    e.preventDefault();
    const form = e.target;
    const submitBtn = form.querySelector('button[type="submit"]');
    submitBtn.innerHTML = `<span class="material-symbols-outlined animate-spin text-[18px]">refresh</span> Envoi en cours...`;
    submitBtn.disabled = true;

    setTimeout(() => {
      form.innerHTML = `
        <div class="text-center py-space-md">
          <div class="w-12 h-12 bg-badge-green-bg text-secondary rounded-full flex items-center justify-center mx-auto mb-space-sm">
            <span class="material-symbols-outlined text-[28px]">thumb_up</span>
          </div>
          <h4 class="font-headline-sm text-headline-sm text-primary font-bold mb-1">Candidature Reçue !</h4>
          <p class="font-body-sm text-body-sm text-on-surface-variant">Le coordinateur régional VoiCI-EMI vous contactera sous 48h avec le kit de déploiement.</p>
        </div>
      `;
      showToast('Dossier enregistré', 'Votre proposition d’atelier a été transmise au pôle territorial.');
    }, 900);
  }

  function confirmModalAction() {
    closeModal();
    showToast('Réservation validée !', 'Un rappel vous sera envoyé 1h avant le début de la session.');
  }

  function closeModal() {
    document.getElementById('quickview-modal').classList.add('hidden');
  }

  // Toast Functionality
  function showToast(title, desc) {
    const toast = document.getElementById('toast');
    document.getElementById('toast-title').innerText = title;
    document.getElementById('toast-desc').innerText = desc;
    toast.classList.remove('translate-y-32', 'opacity-0');
    toast.classList.add('translate-y-0', 'opacity-100');

    setTimeout(() => {
      toast.classList.add('translate-y-32', 'opacity-0');
      toast.classList.remove('translate-y-0', 'opacity-100');
    }, 3800);
  }
